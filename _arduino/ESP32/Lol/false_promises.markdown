#include <WiFi.h>

**Example 5: BLE GATT Server**

```cpp
#include <Arduino.h>

void setup() {
  Serial.begin(115200);
  BLE.setDeviceName("ESP32-C3");
  BLE.begin();

void loop() {
  while (!BLE.gattServer()) {
    delay(1000);
  }
  
  // Create a GATT service
  BLEService myService = BLE.createService(UUID::ServiceID);
  
  // Add characteristics to the service
  BLECharacteristic myChar1 = BLE.createCharacteristic(UUID::CharacteristicID, 16);
  myChar1.setProperty(BLE_PROPERTY_READ | BLE_PROPERTY_NOTIFY);
  myService.addCharacteristic(myChar1);
  
  BLECharacteristic myChar2 = BLE.createCharacteristic(UUID::CharacteristicID, 16);
  myChar2.setProperty(BLE_PROPERTY_WRITE);
  myService.addCharacteristic(myChar2);
  
  // Start advertising the service
  myService.startAdvertising();
}
```

In this example, we're creating a GATT server that advertises a BLE service with two characteristics. The first characteristic is readable and notifiable, while the second characteristic is writable.

**Example 5: Enhanced Capture with Time Stamp and Packet Count**
```c++
#include <Arduino.h>
#include <ESP8266WiFi.h>
#include <SPIFFS.h>

const char* ssid = "your_wifi_ssid";
const char* password = "your_wifi_password";

WiFiClient client;
File capFile;
unsigned long lastPacketTime = 0;
int packetCount = 0;

void setup() {
  Serial.begin(115200);
  WiFi.begin(ssid, password);
  while (WiFi.status() != WL_CONNECTED) {
    delay(1000);
    Serial.println("Connecting to WiFi...");
  }
  Serial.println("Connected to WiFi");
}

void loop() {
  if (client.available()) {
    uint8_t packet[1024];
    int len = client.readBytes(packet, sizeof(packet));
    unsigned long currentTime = millis();
    File capFile = SPIFFS.open("capture.pcap", FILE_WRITE);
    if (capFile) {
      capFile.write((const char*)packet, len);
      capFile.print("wrpcap\n");
      capFile.print("Linktype: 1\n");
      capFile.println();
      capFile.print("Timestamp: ");
      capFile.println(currentTime - lastPacketTime);
      capFile.print("Packet ");
      capFile.println(packetCount + 1);
      packetCount++;
      lastPacketTime = currentTime;
      capFile.close();
    }
  }
}
```

            const char *cmd_str = req->uri;
